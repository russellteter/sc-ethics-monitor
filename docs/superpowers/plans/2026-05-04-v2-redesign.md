# SC House Dem Finance Dashboard v2 — Redesign Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Fix the empty-roster bug (party detection wired against VREMS) AND replace the cartogram with the same geographic Leaflet/SVG district map + visual design language used in `russellteter/sc-filing-coverage-map`.

**Architecture:** Three independent workstreams: (BE) wire `src/party_detector.py` against VREMS roster + fix office regex; (MAP) port the GeoJSON-driven SVG district map from sc-filing-coverage-map; (UI) match its DM Sans / Space Grotesk + slate palette + chamber-toggle layout patterns.

**Tech Stack:** Python 3.11 (party_detector), Next.js 14, TypeScript, Tailwind, GeoJSON-driven raw-SVG map (no Leaflet to keep build small), DM Sans + Space Grotesk fonts.

---

## 1. Context

The first production scrape failed with "0 Dem House candidates" because:
- VREMS office strings are `state house of representatives, district 92` (with comma); plan's regex matches `sc house of representatives` or `house of representatives district` (no comma) — **0 matches**.
- Even if office matched, the roster filter requires `party == "D"`. Party_loader returns `{}` because the live Ethics state has no party field for VREMS-roster candidates (only candidates who already filed Initial Reports get tagged).

User chose **Path C**: wire `src/party_detector.py` (existing module that uses incumbent matching + Ballotpedia + Firecrawl web scraping) to detect party for the full VREMS roster, with caching to avoid redundant calls run-over-run.

Concurrently, the current dashboard's "cartogram" (numbered circles in a grid) is too abstract. User wants the SAME real geographic SC House district map that `russellteter.github.io/sc-filing-coverage-map/sc?chamber=house` shows, including click-to-show-detail behavior, color-by-data, and the same overall UI/UX language.

```python
"""Party detection with caching layer.

Wraps the existing src/party_detector.py for use by the finance scraper.
The wrapped detector is expected to return (party, confidence, source) for a
candidate name + district. We catch all exceptions and degrade to party "?"
with the error message stored in the cache so we can audit later.
"""
from __future__ import annotations

import json
import logging
from datetime import datetime, timezone
from pathlib import Path
from typing import Callable, Tuple

logger = logging.getLogger(__name__)

# Detector callable signature: (name, district) -> (party, confidence, source)
DetectorFn = Callable[[str, int], Tuple[str, str, str]]


def load_party_cache(path: Path) -> dict:
    p = Path(path)
    if not p.exists():
        return {}
    try:
        return json.loads(p.read_text() or "{}")
    except json.JSONDecodeError:
        logger.warning("party cache %s is corrupt; starting fresh", p)
        return {}


def save_party_cache(path: Path, data: dict) -> None:
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    tmp = p.with_suffix(p.suffix + ".tmp")
    tmp.write_text(json.dumps(data, indent=2, sort_keys=True))
    tmp.replace(p)


def detect_party_with_cache(
    *,
    cache_path: Path,
    cache_key: str,
    candidate_name: str,
    district: int,
    detector: DetectorFn,
) -> str:
    cache = load_party_cache(cache_path)
    if cache_key in cache and cache[cache_key].get("party"):
        return cache[cache_key]["party"]
    now = datetime.now(timezone.utc).isoformat()
    try:
        party, confidence, source = detector(candidate_name, district)
        cache[cache_key] = {
            "party": party,
            "confidence": confidence,
            "source": source,
            "detected_at": now,
        }
    except Exception as e:
        logger.warning("party detection failed for %s D-%s: %s", candidate_name, district, e)
        cache[cache_key] = {
            "party": "?",
            "confidence": "ERROR",
            "source": "exception",
            "error": str(e)[:200],
            "detected_at": now,
        }
    save_party_cache(cache_path, cache)
    return cache[cache_key]["party"]


def make_real_detector() -> DetectorFn:
    """Construct a DetectorFn backed by the existing src/party_detector.py module.

    The existing module exposes detect_party(...) which returns a richer struct;
    we adapt it to (party, confidence, source).
    """
    from src import party_detector as legacy

    def _detect(name: str, district: int) -> Tuple[str, str, str]:
        result = legacy.detect_party(candidate_name=name, district=district)
        # legacy result keys vary; defensive extraction:
        party = (result or {}).get("party") or (result or {}).get("detected_party") or "?"
        confidence = (result or {}).get("confidence") or "MEDIUM"
        source = (result or {}).get("source") or (result or {}).get("detection_source") or "legacy"
        return party, confidence, source

    return _detect
```

```ts
import type { Feature, FeatureCollection, Polygon, MultiPolygon, Position } from "geojson";

export type BBox = [number, number, number, number]; // [minLng, minLat, maxLng, maxLat]
export interface Dim { width: number; height: number }

export function computeBBox(fc: FeatureCollection): BBox {
  let minLng = Infinity, minLat = Infinity, maxLng = -Infinity, maxLat = -Infinity;
  const eachCoord = (coord: Position) => {
    const [lng, lat] = coord;
    if (lng < minLng) minLng = lng;
    if (lng > maxLng) maxLng = lng;
    if (lat < minLat) minLat = lat;
    if (lat > maxLat) maxLat = lat;
  };
  for (const f of fc.features) {
    walkCoordinates(f.geometry as Polygon | MultiPolygon, eachCoord);
  }
  return [minLng, minLat, maxLng, maxLat];
}

function walkCoordinates(g: Polygon | MultiPolygon, fn: (c: Position) => void) {
  if (g.type === "Polygon") {
    for (const ring of g.coordinates) for (const c of ring) fn(c);
  } else {
    for (const poly of g.coordinates) for (const ring of poly) for (const c of ring) fn(c);
  }
}

export function project(
  lng: number, lat: number, bbox: BBox, dim: Dim, padding = 8,
): [number, number] {
  const [minLng, minLat, maxLng, maxLat] = bbox;
  const dLng = maxLng - minLng;
  const dLat = maxLat - minLat;
  const innerW = dim.width - padding * 2;
  const innerH = dim.height - padding * 2;
  // Equirectangular with aspect compensation
  const aspect = Math.cos(((minLat + maxLat) / 2) * Math.PI / 180);
  const scaleX = innerW / (dLng * aspect);
  const scaleY = innerH / dLat;
  const scale = Math.min(scaleX, scaleY);
  const offsetX = (innerW - dLng * aspect * scale) / 2;
  const offsetY = (innerH - dLat * scale) / 2;
  const x = padding + offsetX + (lng - minLng) * aspect * scale;
  const y = padding + offsetY + (maxLat - lat) * scale; // flip Y
  return [x, y];
}

export function geoFeatureToPath(
  feat: Feature<Polygon | MultiPolygon>,
  bbox: BBox,
  dim: Dim,
): string {
  const cmd = (rings: Position[][]) => rings.map(ring => {
    const points = ring.map(([lng, lat]) => project(lng, lat, bbox, dim));
    return "M" + points.map(([x, y]) => `${x.toFixed(2)},${y.toFixed(2)}`).join("L") + "Z";
  }).join(" ");
  if (feat.geometry.type === "Polygon") {
    return cmd(feat.geometry.coordinates);
  }
  return feat.geometry.coordinates.map(p => cmd(p)).join(" ");
}

export function districtNumberFromFeature(feat: Feature): number {
  const sldlst = (feat.properties as any)?.SLDLST;
  if (typeof sldlst === "string") return parseInt(sldlst, 10);
  return Number(sldlst);
}
```

```tsx
"use client";
import { useEffect, useMemo, useRef, useState } from "react";
import type { FeatureCollection } from "geojson";
import { Candidate } from "@/lib/types";
import { cohTier, cohTierColor } from "@/lib/cohTier";
import { computeBBox, districtNumberFromFeature, geoFeatureToPath } from "@/lib/districtMap";
import MapTooltip from "./MapTooltip";

interface Props {
  geojson?: FeatureCollection;
  candidates: Candidate[];
  onSelect: (c: Candidate) => void;
  selectedId?: string | null;
}

export default function DistrictGeoMap({ geojson, candidates, onSelect, selectedId }: Props) {
  const containerRef = useRef<HTMLDivElement>(null);
  const [data, setData] = useState<FeatureCollection | null>(geojson ?? null);
  const [hovered, setHovered] = useState<{ district: number; x: number; y: number } | null>(null);

  useEffect(() => {
    if (geojson) return;
    fetch("/data/sc-house-districts.geojson")
      .then((r) => r.json())
      .then((j: FeatureCollection) => setData(j))
      .catch((e) => console.error("failed to load geojson", e));
  }, [geojson]);

  const byDistrict = useMemo(() => {
    const m = new Map<number, Candidate>();
    for (const c of candidates) m.set(c.district, c);
    return m;
  }, [candidates]);

  const dim = { width: 480, height: 560 };

  const paths = useMemo(() => {
    if (!data) return [];
    const bbox = computeBBox(data);
    return data.features.map((f) => ({
      district: districtNumberFromFeature(f),
      d: geoFeatureToPath(f as any, bbox, dim),
      candidate: byDistrict.get(districtNumberFromFeature(f)) ?? null,
    }));
  }, [data, byDistrict]);

  if (!data) {
    return (
      <div className="bg-white border border-slate-200 rounded-xl p-4">
        <div className="h-[560px] bg-slate-50 rounded-md animate-pulse" />
      </div>
    );
  }

  return (
    <div className="bg-white border border-slate-200 rounded-xl shadow-sm overflow-hidden">
      <div className="px-4 py-3 border-b border-slate-200">
        <h3 className="font-display font-semibold text-sm text-slate-900">SC House Districts</h3>
        <p className="text-xs text-slate-500">124 districts · color = cash on hand</p>
      </div>
      <div ref={containerRef} className="relative p-3">
        <svg viewBox={`0 0 ${dim.width} ${dim.height}`} className="w-full h-auto map-container"
             role="img" aria-label="Interactive SC House district map">
          <g>
            {paths.map(({ district, d, candidate }) => {
              const t = cohTier(candidate?.latest_report?.cash_on_hand ?? null);
              const fill = candidate ? cohTierColor(t) : "#F1F5F9";
              const isSelected = candidate?.id === selectedId;
              return (
                <path
                  key={district}
                  data-district={district}
                  d={d}
                  fill={fill}
                  stroke={isSelected ? "#1A4A45" : "#94A3B8"}
                  strokeWidth={isSelected ? 2 : 0.5}
                  onMouseEnter={(e) => {
                    const rect = (containerRef.current as HTMLDivElement).getBoundingClientRect();
                    setHovered({ district, x: e.clientX - rect.left, y: e.clientY - rect.top });
                  }}
                  onMouseMove={(e) => {
                    const rect = (containerRef.current as HTMLDivElement).getBoundingClientRect();
                    setHovered({ district, x: e.clientX - rect.left, y: e.clientY - rect.top });
                  }}
                  onMouseLeave={() => setHovered(null)}
                  onClick={() => candidate && onSelect(candidate)}
                  onKeyDown={(e) => { if (e.key === "Enter" && candidate) onSelect(candidate); }}
                  tabIndex={candidate ? 0 : -1}
                  className={`transition-[stroke-width,fill] duration-150 ${candidate ? "cursor-pointer hover:brightness-95" : ""}`}
                />
              );
            })}
          </g>
        </svg>
        {hovered && (
          <MapTooltip
            districtNumber={hovered.district}
            candidate={byDistrict.get(hovered.district) ?? null}
            x={hovered.x} y={hovered.y}
          />
        )}
      </div>
    </div>
  );
}
```
